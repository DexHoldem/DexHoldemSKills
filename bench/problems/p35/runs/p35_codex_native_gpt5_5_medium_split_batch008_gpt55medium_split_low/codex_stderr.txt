        return False

def normalize_steps(raw):
    steps = []
    for item in raw or []:
        if isinstance(item, dict):
            name = item.get("name")
            if not name:
                raise RuntimeError("sequence step is missing name")
            step = dict(item)
            step.setdefault("status", "pending")
        else:
            step = {"name": str(item), "status": "pending"}
        steps.append(step)
    return steps
