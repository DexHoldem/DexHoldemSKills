    path = Path(path)
    if not path.exists():

    try:
        blocks = extract_json_objects(path.read_text())
    except OSError as exc:

    return None, f"{path.name} has no matching JSON block"

def normalize_steps(raw):

    for item in raw or []:
        if isinstance(item, dict):
            name = item.get("name")

            step = dict(item)
            step.setdefault("status", "pending")
        else:
